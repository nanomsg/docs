Survey (Everybody Votes)

The surveyor pattern is used to send a timed survey out, responses are individually returned until the survey has expired. This pattern is useful for service discovery and voting algorithms.

survey.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <unistd.h>

#include <nng/nng.h>

static nng_err
nng_recv_alloc_(nng_socket sock, void **buf, size_t *size)
{
        nng_msg *msg = NULL;
        nng_err rv;
        if ((rv = nng_recvmsg(sock, &msg, 0)) != NNG_OK) return rv;
        *size = nng_msg_len(msg);
        if ((*buf = nng_alloc(*size)) == NULL) { nng_msg_free(msg); return NNG_ENOMEM; }
        memcpy(*buf, nng_msg_body(msg), *size);
        nng_msg_free(msg);
        return NNG_OK;
}
#define nng_recv_alloc(sock, buf, size) nng_recv_alloc_((sock), (void **)(buf), (size))

#define SERVER "server"
#define CLIENT "client"
#define DATE   "DATE"

void
fatal(const char *func, nng_err rv)
{
        fprintf(stderr, "%s: %s\n", func, nng_strerror(rv));
        exit(1);
}

char *
date(void)
{
        time_t now = time(&now);
        struct tm *info = localtime(&now);
        char *text = asctime(info);
        text[strlen(text)-1] = '\0'; // remove '\n'
        return (text);
}

int
server(const char *url)
{
        nng_socket sock;
	nng_err rv;

        if ((rv = nng_surveyor0_open(&sock)) != NNG_OK) {
                fatal("nng_surveyor0_open", rv);
        }
        if ((rv = nng_listen(sock, url, NULL, 0)) != NNG_OK) {
                fatal("nng_listen", rv);
        }
        for (;;) {
                printf("SERVER: SENDING DATE SURVEY REQUEST\n");
                if ((rv = nng_send(sock, DATE, strlen(DATE) + 1, 0)) != NNG_OK) {
                        fatal("nng_send", rv);
                }

                for (;;) {
                        char *buf = NULL;
			size_t sz;
                        rv = nng_recv_alloc(sock, &buf, &sz);
			if (rv == NNG_ETIMEDOUT) {
				break;
			}
			if (rv != NNG_OK) {
				fatal("nng_recv", rv);
			}
                        printf("SERVER: RECEIVED \"%s\" SURVEY RESPONSE\n",
                            buf); (1)
                        nng_free(buf, sz);
                }

                printf("SERVER: SURVEY COMPLETE\n");
        }
}

int
client(const char *url, const char *name)
{
        nng_socket sock;
	nng_err rv;

        if ((rv = nng_respondent0_open(&sock)) != NNG_OK) {
                fatal("nng_respondent0_open", rv);
        }
        if ((rv = nng_dial(sock, url, NULL, NNG_FLAG_NONBLOCK)) != NNG_OK) {
                fatal("nng_dial", rv);
        }
	for (;;) {
                char *buf = NULL;
		size_t sz;
                if ((rv = nng_recv_alloc(sock, &buf, &sz)) == NNG_OK) {
                        printf("CLIENT (%s): RECEIVED \"%s\" SURVEY REQUEST\n",
                            name, buf); (1)
                        nng_free(buf, sz);
                        char *d = date();
                        printf("CLIENT (%s): SENDING DATE SURVEY RESPONSE\n",
                           name);
                        if ((rv = nng_send(sock, d, strlen(d) + 1, 0)) != NNG_OK) {
                                fatal("nng_send", rv);
                        }
                }
        }
}

int
main(const int argc, const char **argv)
{
        if ((argc >= 2) && (strcmp(SERVER, argv[1]) == 0))
                return (server(argv[2]));

        if ((argc >= 3) && (strcmp(CLIENT, argv[1]) == 0))
                return (client(argv[2], argv[3]));

        fprintf(stderr, "Usage: survey %s|%s <URL> <ARG> ...\n",
            SERVER, CLIENT);
        return 1;
}

Compilation
gcc survey.c -lnng -o survey
Execution
./survey server ipc:///tmp/survey.ipc & server=$!
./survey client ipc:///tmp/survey.ipc client0 & client0=$!
./survey client ipc:///tmp/survey.ipc client1 & client1=$!
./survey client ipc:///tmp/survey.ipc client2 & client2=$!
sleep 3 (1)
kill $server $client0 $client1 $client2
