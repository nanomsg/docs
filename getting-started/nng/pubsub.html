Pub/Sub (Topics & Broadcast)

This pattern is used to allow a single broadcaster to publish messages to many subscribers, which may choose to limit which messages they receive.

pubsub.c
#include <stdlib.h>
#include <stdio.h>
#include <string.h>
#include <time.h>
#include <unistd.h>

#include <nng/nng.h>

static nng_err
nng_recv_alloc_(nng_socket sock, void **buf, size_t *size)
{
        nng_msg *msg = NULL;
        nng_err rv;
        if ((rv = nng_recvmsg(sock, &msg, 0)) != NNG_OK) return rv;
        *size = nng_msg_len(msg);
        if ((*buf = nng_alloc(*size)) == NULL) { nng_msg_free(msg); return NNG_ENOMEM; }
        memcpy(*buf, nng_msg_body(msg), *size);
        nng_msg_free(msg);
        return NNG_OK;
}
#define nng_recv_alloc(sock, buf, size) nng_recv_alloc_((sock), (void **)(buf), (size))

#define SERVER "server"
#define CLIENT "client"

void
fatal(const char *func, nng_err rv)
{
        fprintf(stderr, "%s: %s\n", func, nng_strerror(rv));
}

char *
date(void)
{
        time_t now = time(&now);
        struct tm *info = localtime(&now);
        char *text = asctime(info);
        text[strlen(text)-1] = '\0'; // remove '\n'
        return (text);
}

int
server(const char *url)
{
        nng_socket sock;
	nng_err rv;

        if ((rv = nng_pub0_open(&sock)) != NNG_OK) {
                fatal("nng_pub0_open", rv);
        }
	if ((rv = nng_listen(sock, url, NULL, 0)) != NNG_OK) {
                fatal("nng_listen", rv);
        }
        for (;;) {
                char *d = date();
                printf("SERVER: PUBLISHING DATE %s\n", d);
                if ((rv = nng_send(sock, d, strlen(d) + 1, 0)) != NNG_OK) {
                        fatal("nng_send", rv);
                }
                sleep(1);
        }
}

int
client(const char *url, const char *name)
{
        nng_socket sock;
	nng_err rv;

        if ((rv = nng_sub0_open(&sock)) != NNG_OK) {
                fatal("nng_sub0_open", rv);
        }

        // subscribe to everything (empty means all topics)
        if ((rv = nng_sub0_socket_subscribe(sock, "", 0)) != NNG_OK) {
                fatal("nng_sub0_socket_subscribe", rv);
        }
        if ((rv = nng_dial(sock, url, NULL, 0)) != NNG_OK) {
                fatal("nng_dial", rv);
        }
        for (;;) {
                char *buf = NULL;
		size_t sz;
                if ((rv = nng_recv_alloc(sock, &buf, &sz)) != NNG_OK) {
                        fatal("nng_recv", rv);
                }
                printf("CLIENT (%s): RECEIVED %s\n", name, buf); (1)
                nng_free(buf, sz);
        }
}

int
main(const int argc, const char **argv)
{
        if ((argc >= 2) && (strcmp(SERVER, argv[1]) == 0))
                return (server(argv[2]));

          if ((argc >= 3) && (strcmp(CLIENT, argv[1]) == 0))
                return (client (argv[2], argv[3]));

        fprintf(stderr, "Usage: pubsub %s|%s <URL> <ARG> ...\n",
            SERVER, CLIENT);
        return 1;
}
1 Blithely assumes message is ASCIIZ string. Real code should check it.
Compilation
gcc pubsub.c -lnng -o pubsub
Execution
./pubsub server ipc:///tmp/pubsub.ipc & server=$! && sleep 1
./pubsub client ipc:///tmp/pubsub.ipc client0 & client0=$!
./pubsub client ipc:///tmp/pubsub.ipc client1 & client1=$!
./pubsub client ipc:///tmp/pubsub.ipc client2 & client2=$!
sleep 5
kill $server $client0 $client1 $client2
