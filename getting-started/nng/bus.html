Bus (Routing)

The bus protocol is useful for routing applications, or for building fully interconnected mesh networks. In this pattern, messages are sent to every directly connected peer.

bus.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>

#include <nng/nng.h>

static nng_err
nng_recv_alloc_(nng_socket sock, void **buf, size_t *size)
{
        nng_msg *msg = NULL;
        nng_err rv;
        if ((rv = nng_recvmsg(sock, &msg, 0)) != NNG_OK) return rv;
        *size = nng_msg_len(msg);
        if ((*buf = nng_alloc(*size)) == NULL) { nng_msg_free(msg); return NNG_ENOMEM; }
        memcpy(*buf, nng_msg_body(msg), *size);
        nng_msg_free(msg);
        return NNG_OK;
}
#define nng_recv_alloc(sock, buf, size) nng_recv_alloc_((sock), (void **)(buf), (size))

void
fatal(const char *func, nng_err rv)
{
        fprintf(stderr, "%s: %s\n", func, nng_strerror(rv));
        exit(1);
}

int
node(int argc, char **argv)
{
        nng_socket sock;
	nng_err rv;
	size_t sz;

        if ((rv = nng_bus0_open(&sock)) != NNG_OK) {
                fatal("nng_bus0_open", rv);
        }
	if ((rv = nng_listen(sock, argv[2], NULL, 0)) != NNG_OK) {
                fatal("nng_listen", rv);
        }

        sleep(1); // wait for peers to bind
        if (argc >= 3) {
                for (int x = 3; x < argc; x++) {
                        if ((rv = nng_dial(sock, argv[x], NULL, 0)) != NNG_OK) {
                                fatal("nng_dial", rv);
                        }
                }
        }

	sleep(1); // wait for connects to establish

        // SEND
        sz = strlen(argv[1]) + 1; // '\0' too
        printf("%s: SENDING '%s' ONTO BUS\n", argv[1], argv[1]);
        if ((rv = nng_send(sock, argv[1], sz, 0)) != NNG_OK) {
                fatal("nng_send", rv);
        }

        // RECV
        for (;;) {
                char *buf = NULL;
		size_t sz;
                if ((rv = nng_recv_alloc(sock, &buf, &sz)) != NNG_OK) {
			if (rv == NNG_ETIMEDOUT) {
				fatal("nng_recv", rv);
			}
		}
		printf("%s: RECEIVED '%s' FROM BUS\n", argv[1], buf); (1)
                nng_free(buf, sz);
        }
        nng_socket_close(sock);
	return (0);
}

int
main(int argc, char **argv)
{
        if (argc >= 3) {
                return (node(argc, argv));
        }
        fprintf(stderr, "Usage: bus <NODE_NAME> <URL> <URL> ...\n");
        return 1;
}
1 Blithely assumes message is ASCIIZ string. Real code should check it.
Compilation
gcc bus.c -lnng -o bus
Execution
./bus node0 ipc:///tmp/node0.ipc ipc:///tmp/node1.ipc ipc:///tmp/node2.ipc & node0=$!
./bus node1 ipc:///tmp/node1.ipc ipc:///tmp/node2.ipc ipc:///tmp/node3.ipc & node1=$!
./bus node2 ipc:///tmp/node2.ipc ipc:///tmp/node3.ipc & node2=$!
./bus node3 ipc:///tmp/node3.ipc ipc:///tmp/node0.ipc & node3=$!
sleep 5
kill $node0 $node1 $node2 $node3
